Ques 44: Write a function SumPdivisor() that finds sum of Proper divisor of a number.
Ques 45:Using this SumPdivisor() ,Write a program to print all the Perfect Number in a given range.
/***********************************************************************************************************/
Code: 

#include <stdio.h>

int SumPdivisor(int n){
    int result, sum = 0;
    for(int i = 1; i < n; i++){
        result = n % i;
        if(result == 0){
            sum = sum + i;
        }
    }
    return sum;
}

int PerfectNumbers(int start, int end){
    int count = 0;
    for(int n = start; n <= end; n++){
        if(n > 0 && SumPdivisor(n) == n){
            printf("perfect number %d\n", n);
            count++;
        }
    }
    return count;
}

int main()
{
    int start, end, ans1;

    printf("Enter range of the number - start num: \n");
    scanf("%d", &start);
    printf("Enter range of the number - end num: \n");
    scanf("%d", &end);

    printf("Perfect numbers are:\n");
    ans1 = PerfectNumbers(start, end);
    printf("Count of perfect numbers found: %d\n", ans1);

    return 0;
}
